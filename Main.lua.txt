----------------------------------------------------------------
local function makeDraggable(f)
    local dragging, dragInput, dragStart, startPos
    f.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = f.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    f.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            local goal = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            TweenService:Create(f, TweenInfo.new(0.1, Enum.EasingStyle.Linear), {Position = goal}):Play()
        end
    end)
end

local function CreateJumpRing()
    if not LP.Character or not LP.Character:FindFirstChild("HumanoidRootPart") then return end
    local root = LP.Character.HumanoidRootPart
    
    local ring = Instance.new("Part")
    ring.Name = "HolyRing"
    ring.Parent = workspace
    ring.Anchored = true
    ring.CanCollide = false
    ring.CastShadow = false
    ring.Material = Enum.Material.Neon
    ring.Color = Data.ThemeColor
    ring.Shape = Enum.PartType.Cylinder
    ring.Size = Vector3.new(0.2, 2, 2)
    ring.CFrame = root.CFrame * CFrame.new(0, -2.5, 0) * CFrame.Angles(0, 0, math.rad(90))
    
    local tween = TweenService:Create(ring, TweenInfo.new(0.6, Enum.EasingStyle.QuartOut), {
        Size = Vector3.new(0.1, 15, 15),
        Transparency = 1
    })
    tween:Play()
    tween.Completed:Connect(function() ring:Destroy() end)
end
